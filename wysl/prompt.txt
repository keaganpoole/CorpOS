export default function App() {
  const [currentSlide, setCurrentSlide] = useState(0);
  const [isTransitioning, setIsTransitioning] = useState(false);
  const [studioLightIntensity, setStudioLightIntensity] = useState(40);
  const [isAutoplay, setIsAutoplay] = useState(true);
  
  const TOTAL_SLIDES = 3;

  useEffect(() => {
    if (!isAutoplay) return;
    const interval = setInterval(() => {
      handleNext();
    }, 7000); 
    return () => clearInterval(interval);
  }, [isAutoplay, currentSlide]);

  const handleNext = () => {
    if (isTransitioning) return;
    setIsTransitioning(true);
    // Increased timeout to allow for a beautiful long fade-out before the next component mounts
    setTimeout(() => {
      setCurrentSlide((prev) => (prev + 1) % TOTAL_SLIDES);
      setIsTransitioning(false);
    }, 600); 
  };

  const handlePrev = () => {
    if (isTransitioning) return;
    setIsTransitioning(true);
    setTimeout(() => {
      setCurrentSlide((prev) => (prev - 1 + TOTAL_SLIDES) % TOTAL_SLIDES);
      setIsTransitioning(false);
    }, 600);
  };

  return (
    <div className="w-full min-h-screen bg-[#000000] text-white antialiased overflow-x-hidden relative flex flex-col justify-between custom-easing duration-700">
      <style>{CSS_ANIMATIONS}</style>
      
      {/* Monochromatic Studio Spotlight Background */}
      <div className="absolute inset-0 overflow-hidden pointer-events-none z-0">
        <div 
          style={{
            opacity: studioLightIntensity / 100 * 0.25,
            background: 'radial-gradient(circle at 50% 40%, rgba(255, 255, 255, 0.4) 0%, rgba(0, 0, 0, 0) 70%)',
            transition: 'opacity 1000ms ease-out'
          }}
          className="absolute inset-0 w-full h-full animate-pulse-glow" 
        />
        <div className="absolute top-0 left-0 w-full h-full bg-[radial-gradient(ellipse_at_top,_var(--tw-gradient-stops))] from-zinc-900/40 via-black to-black opacity-60" />
      </div>

      <main className="flex-grow flex-1 relative z-10 flex flex-col items-center justify-center px-4 md:px-12 py-16 max-w-5xl mx-auto w-full select-none text-center">
        
        {/* Anchored Brand Text Block */}
        <div className="flex flex-col items-center space-y-6 mb-12 max-w-3xl">
          <h1 className="hero-title-match text-4xl md:text-6xl lg:text-7xl text-transparent bg-clip-text bg-gradient-to-b from-white via-zinc-200 to-zinc-600 mb-2">
            automated appointment<br/>
            booking.
          </h1>
          
          <p className="hero-subheadline-match text-lg md:text-xl text-zinc-400 max-w-2xl leading-relaxed">
            Chronos is an elite scheduling canvas engineered for luxury salons. Elevate client hospitality with seamless booking, visual station allocation, and a flawless guest journey.
          </p>

          <div className="pt-2 flex flex-wrap justify-center gap-6">
            <span className="flex items-center space-x-2 text-[10px] font-bold text-zinc-400 tracking-widest uppercase">
              <span className="w-1.5 h-1.5 rounded-full bg-cyan-400 shadow-[0_0_8px_#06b6d4]" />
              <span>Guest Journey</span>
            </span>
            <span className="flex items-center space-x-2 text-[10px] font-bold text-zinc-400 tracking-widest uppercase">
              <span className="w-1.5 h-1.5 rounded-full bg-violet-400 shadow-[0_0_8px_#8b5cf6]" />
              <span>Station Allocation</span>
            </span>
            <span className="flex items-center space-x-2 text-[10px] font-bold text-zinc-400 tracking-widest uppercase">
              <span className="w-1.5 h-1.5 rounded-full bg-rose-400 shadow-[0_0_8px_#f43f5e]" />
              <span>Stylist Harmony</span>
            </span>
          </div>
        </div>

        {/* Dynamic Interactive Showcases with ultra-smooth cinematic transition */}
        <div className="w-full flex items-center justify-center relative min-h-[520px]">
          <div 
            className={`w-full premium-easing ${
              isTransitioning ? 'canvas-exit' : 'canvas-enter'
            }`}
          >
            {currentSlide === 0 && <RightHeroWidget />}
            {currentSlide === 1 && <RightCalendarGrid />}
            {currentSlide === 2 && <RightNlpSandbox />}
          </div>
        </div>

      </main>

      {/* Footer Navigation */}
      <footer className="relative z-50 px-8 py-8 max-w-7xl mx-auto w-full flex flex-col md:flex-row items-center justify-between gap-6 border-t border-white/5">
        <div className="flex items-center space-x-3">
          {[0, 1, 2].map((idx) => {
            const activeColors = [
              'bg-gradient-to-r from-cyan-400 to-blue-500 shadow-[0_0_15px_rgba(6,182,212,0.5)]',
              'bg-gradient-to-r from-violet-500 to-fuchsia-500 shadow-[0_0_15px_rgba(139,92,246,0.5)]',
              'bg-gradient-to-r from-rose-500 to-amber-500 shadow-[0_0_15px_rgba(244,63,94,0.5)]'
            ];
            
            return (
              <button
                key={idx}
                onClick={() => {
                  if (isTransitioning) return;
                  setIsTransitioning(true);
                  setTimeout(() => {
                    setCurrentSlide(idx);
                    setIsTransitioning(false);
                  }, 600);
                }}
                className="group relative py-2"
              >
                <div className={`h-1.5 rounded-full transition-all duration-700 ${currentSlide === idx ? `w-20 ${activeColors[idx]}` : 'w-8 bg-zinc-800 hover:bg-zinc-600'}`} />
              </button>
            );
          })}
        </div>

        <div className="flex items-center space-x-6">
          <button 
            onClick={() => setIsAutoplay(!isAutoplay)}
            className={`px-3 py-1 rounded-full text-[9px] font-bold tracking-widest border transition-all duration-500 ${isAutoplay ? 'bg-white text-black border-white' : 'border-zinc-800 text-zinc-500'}`}
          >
            {isAutoplay ? 'AUTOPLAY ACTIVE' : 'PAUSED'}
          </button>

          <div className="flex items-center space-x-3">
            <button
              onClick={handlePrev}
              className="w-12 h-12 rounded-full border border-zinc-800 bg-zinc-950/40 text-white flex items-center justify-center hover:bg-zinc-900 hover:border-zinc-700 transition-all duration-300 hover:scale-105 active:scale-95"
            >
              <ChevronLeft size={18} />
            </button>
            <button
              onClick={handleNext}
              className="w-28 h-12 rounded-full bg-white text-black flex items-center justify-center gap-2 font-semibold transition-all duration-300 hover:scale-105 active:scale-95 hover:bg-zinc-200"
            >
              <span>Next</span>
              <ArrowRight size={14} />
            </button>
          </div>
        </div>
      </footer>
    </div>
  );
}

function RightHeroWidget() {
  const [hoverPosition, setHoverPosition] = useState({ x: 0, y: 0 });
  const containerRef = useRef(null);

  const handleMouseMove = (e) => {
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const x = (e.clientX - rect.left) / rect.width - 0.5;
    const y = (e.clientY - rect.top) / rect.height - 0.5;
    setHoverPosition({ x, y });
  };

  const handleMouseLeave = () => {
    setHoverPosition({ x: 0, y: 0 });
  };

  const scheduledItems = [
    { time: "09:00 AM", title: "Balayage & Couture Blowout", detail: "Marcus - VIP Salon Suite", colorClass: "from-cyan-500 to-blue-500", tag: "Color", tagColor: "bg-cyan-500/10 text-cyan-400 border-cyan-500/20" },

  ];

  return (
    <div ref={containerRef} onMouseMove={handleMouseMove} onMouseLeave={handleMouseLeave} className="w-full flex justify-center py-4">
      <div style={{ transform: `perspective(1000px) rotateY(${hoverPosition.x * 12}deg) rotateX(${-hoverPosition.y * 12}deg) scale(1.02)`, transition: 'transform 800ms cubic-bezier(0.2, 0.8, 0.2, 1)', boxShadow: '0 30px 60px -15px rgba(0,0,0,0.9), 0 0 40px -20px rgba(6,182,212,0.3)' }} className="w-full max-w-md aspect-[4/5] rounded-[24px] border border-white/10 bg-gradient-to-b from-zinc-950 to-[#030303] p-8 relative overflow-hidden group transition-all">

        </div>
        <div className="space-y-3.5 text-left">
