import React, { useState, useEffect, useRef } from 'react';

// ==========================================
// UTILS & SHARED LOGIC
// ==========================================
const useCanvasResize = (canvasRef, containerRef) => {
  const [size, setSize] = useState({ width: 0, height: 0 });

  useEffect(() => {
    const handleResize = () => {
      const canvas = canvasRef.current;
      const container = containerRef.current;
      if (!canvas || !container) return;

      const rect = container.getBoundingClientRect();
      const dpr = window.devicePixelRatio || 1;
      
      canvas.width = rect.width * dpr;
      canvas.height = rect.height * dpr;
      canvas.style.width = `${rect.width}px`;
      canvas.style.height = `${rect.height}px`;

      setSize({ width: rect.width, height: rect.height });
    };

    handleResize();
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, [canvasRef, containerRef]);

  return size;
};

// ==========================================
// METAMORPHIC FLUID (The Organic Blob)
// ==========================================
const MetamorphicFluid = ({ isExpanded }) => {
  const canvasRef = useRef(null);
  const containerRef = useRef(null);
  const size = useCanvasResize(canvasRef, containerRef);
  const mouseRef = useRef({ x: 0, y: 0, tx: 0, ty: 0 });

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    let animationId;
    let time = 0;

    const render = () => {
      const dpr = window.devicePixelRatio || 1;
      const w = canvas.width;
      const h = canvas.height;
      const cx = w / 2;
      const cy = h / 2;

      ctx.fillStyle = '#030108';
      ctx.fillRect(0, 0, w, h);

      time += 0.008;
      const mouse = mouseRef.current;
      mouse.x += (mouse.tx - mouse.x) * 0.08;
      mouse.y += (mouse.ty - mouse.y) * 0.08;

      const baseRad = Math.min(w, h) * (isExpanded ? 0.28 : 0.22);

      // Helper to generate fluid blob coordinates
      const getBlobCoords = (rad, speedScale, offsetScale, noiseFreq) => {
        const coords = [];
        const steps = 180;
        for (let i = 0; i <= steps; i++) {
          const angle = (i / steps) * Math.PI * 2;
          
          // Synthesizing fluid noise using layered sines
          const wave1 = Math.sin(angle * 3 + time * speedScale) * 10;
          const wave2 = Math.cos(angle * 5 - time * 1.4 * speedScale) * 12;
          const wave3 = Math.sin(angle * 8 + time * 0.7) * 6;
          
          // Interactive morph warp based on mouse
          const mouseWarp = Math.cos(angle - Math.atan2(mouse.y, mouse.x)) * (Math.sqrt(mouse.x*mouse.x + mouse.y*mouse.y) * 45);

          const r = rad + (wave1 + wave2 + wave3) * offsetScale + mouseWarp;
          const px = cx + Math.cos(angle) * r;
          const py = cy + Math.sin(angle) * r;
          coords.push({ x: px, y: py });
        }
        return coords;
      };

      // 1. Giant Soft Gradient Aura Outer Ring
      const outerCoords = getBlobCoords(baseRad * 1.35, 0.7, 1.8, 1);
      ctx.save();
      ctx.beginPath();
      outerCoords.forEach((p, idx) => {
        if (idx === 0) ctx.moveTo(p.x, p.y);
        else ctx.lineTo(p.x, p.y);
      });
      ctx.closePath();
      
      const auraGrad = ctx.createRadialGradient(cx, cy, baseRad * 0.5, cx, cy, baseRad * 1.8);
      auraGrad.addColorStop(0, 'rgba(236, 72, 153, 0.15)');
      auraGrad.addColorStop(0.5, 'rgba(168, 85, 247, 0.06)');
      auraGrad.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = auraGrad;
      ctx.fill();
      ctx.restore();

      // 2. Translucent Middle Core Ring
      const midCoords = getBlobCoords(baseRad * 1.05, 1.2, 1.2, 1.5);
      ctx.save();
      ctx.beginPath();
      midCoords.forEach((p, idx) => {
        if (idx === 0) ctx.moveTo(p.x, p.y);
        else ctx.lineTo(p.x, p.y);
      });
      ctx.closePath();
      const midGrad = ctx.createLinearGradient(0, 0, w, h);
      midGrad.addColorStop(0, 'rgba(236, 72, 153, 0.2)');
      midGrad.addColorStop(1, 'rgba(168, 85, 247, 0.25)');
      ctx.strokeStyle = 'rgba(236, 72, 153, 0.4)';
      ctx.lineWidth = 1 * dpr;
      ctx.stroke();
      ctx.fillStyle = midGrad;
      ctx.fill();
      ctx.restore();

      // 3. Central Solid Morphing Core
      const coreCoords = getBlobCoords(baseRad * 0.85, 1.5, 0.8, 2);
      ctx.save();
      ctx.beginPath();
      coreCoords.forEach((p, idx) => {
        if (idx === 0) ctx.moveTo(p.x, p.y);
        else ctx.lineTo(p.x, p.y);
      });
      ctx.closePath();

      const coreGrad = ctx.createLinearGradient(cx - baseRad, cy - baseRad, cx + baseRad, cy + baseRad);
      coreGrad.addColorStop(0, '#ec4899'); // Neon Pink
      coreGrad.addColorStop(1, '#6b21a8'); // Ultra Violet Deep
      
      ctx.fillStyle = coreGrad;
      ctx.shadowColor = 'rgba(236, 72, 153, 0.5)';
      ctx.shadowBlur = 35 * dpr;
      ctx.fill();
      ctx.restore();

      animationId = requestAnimationFrame(render);
    };

    render();

    const handleMouseMove = (e) => {
      const rect = canvas.getBoundingClientRect();
      const x = (e.clientX - rect.left) / rect.width - 0.5;
      const y = (e.clientY - rect.top) / rect.height - 0.5;
      mouseRef.current.tx = x;
      mouseRef.current.ty = y;
    };

    const handleMouseLeave = () => {
      mouseRef.current.tx = 0;
      mouseRef.current.ty = 0;
    };

    canvas.addEventListener('mousemove', handleMouseMove);
    canvas.addEventListener('mouseleave', handleMouseLeave);

    return () => {
      cancelAnimationFrame(animationId);
      canvas.removeEventListener('mousemove', handleMouseMove);
      canvas.removeEventListener('mouseleave', handleMouseLeave);
    };
  }, [size, isExpanded]);

  return (
    <div ref={containerRef} className="w-full h-full relative overflow-hidden bg-[#030108]">
      <canvas ref={canvasRef} className="block w-full h-full" />
      <div className="absolute bottom-6 left-6 font-mono text-[10px] tracking-[0.2em] text-purple-400/50 uppercase select-none pointer-events-none">
        02 // METAMORPHIC FLUID
      </div>
    </div>
  );
};

// ==========================================
// MAIN APP COMPONENT
// ==========================================
export default function App() {
  return (
    <div className="min-h-screen w-full bg-[#020005] overflow-hidden select-none flex flex-col justify-between font-sans relative">
      
      {/* Absolute faint background ambient noise grid */}
      <div className="absolute inset-0 bg-[linear-gradient(to_right,rgba(168,85,247,0.02)_1px,transparent_1px),linear-gradient(to_bottom,rgba(168,85,247,0.02)_1px,transparent_1px)] bg-[size:4rem_4rem] pointer-events-none z-0" />

      {/* Main Single Presentation Layer */}
      <div className="flex-grow w-full h-full relative z-10">
        <MetamorphicFluid isExpanded={true} />
      </div>

    </div>
  );
}
