import React, { useState, useRef, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import {
  Palette, Type, Sparkles, Check, RotateCcw, Trash2, X,
  ChevronDown, Plus, Hash, HelpCircle, GripVertical
} from 'lucide-react';

// The exact 18 colors from your original configuration
const OPTION_COLORS = [
  '#10b981', '#06b6d4', '#3b82f6', '#6366f1', '#8b5cf6',
  '#a855f7', '#d946ef', '#ec4899', '#f43f5e', '#ef4444',
  '#f97316', '#f59e0b', '#eab308', '#84cc16', '#22c55e',
  '#71717a', '#a1a1aa', '#d4d4d8',
];

// Helper to convert hex to RGBA for soft tag backgrounds
const hexToRgba = (hex, alpha = 0.15) => {
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  return `rgba(${r}, ${g}, ${b}, ${alpha})`;
};

export default function App() {
  // Mock states to simulate the modal tab structure
  const [activeTab, setActiveTab] = useState('colors');
  const [saved, setSaved] = useState(false);

  // Dynamic list of options for live-testing
  const [options, setOptions] = useState([
    'High Priority',
    'Medium Priority',
    'Low Priority',
    'In Progress',
    'Completed',
    'On Hold'
  ]);

  // Initial color mapping
  const [optionColors, setOptionColors] = useState({
    'High Priority': '#f43f5e',
    'Medium Priority': '#f59e0b',
    'Low Priority': '#3b82f6',
    'In Progress': '#06b6d4',
    'Completed': '#10b981',
    'On Hold': '#71717a'
  });

  // Controls which option's color popover is currently open
  const [activePopover, setActivePopover] = useState(null); // stores option name
  const [newOptionName, setNewOptionName] = useState('');

  const handleColorChange = (option, color) => {
    setOptionColors(prev => ({
      ...prev,
      [option]: color
    }));
  };

  const handleAddOption = (e) => {
    e.preventDefault();
    if (!newOptionName.trim()) return;
    const name = newOptionName.trim();
    if (!options.includes(name)) {
      setOptions(prev => [...prev, name]);
      // Assign a random color from the list for fun
      const randomColor = OPTION_COLORS[Math.floor(Math.random() * OPTION_COLORS.length)];
      setOptionColors(prev => ({ ...prev, [name]: randomColor }));
    }
    setNewOptionName('');
  };

  const handleRemoveOption = (name) => {
    setOptions(prev => prev.filter(o => o !== name));
    const nextColors = { ...optionColors };
    delete nextColors[name];
    setOptionColors(nextColors);
    if (activePopover === name) setActivePopover(null);
  };

  const handleReset = () => {
    setOptions(['High Priority', 'Medium Priority', 'Low Priority', 'In Progress', 'Completed', 'On Hold']);
    setOptionColors({
      'High Priority': '#f43f5e',
      'Medium Priority': '#f59e0b',
      'Low Priority': '#3b82f6',
      'In Progress': '#06b6d4',
      'Completed': '#10b981',
      'On Hold': '#71717a'
    });
    setActivePopover(null);
  };

  const handleSave = () => {
    setSaved(true);
    setTimeout(() => setSaved(false), 1500);
  };

  return (
    <div className="min-h-screen bg-[#09090b] flex flex-col items-center justify-center p-4 selection:bg-indigo-500/30 selection:text-white">
      
      {/* Visual background decorations to mimic a rich app context */}
      <div className="absolute top-1/4 left-1/4 w-96 h-96 bg-indigo-500/10 rounded-full blur-[120px] pointer-events-none" />
      <div className="absolute bottom-1/4 right-1/4 w-96 h-96 bg-cyan-500/10 rounded-full blur-[120px] pointer-events-none" />

      {/* Header Info Banner */}
      <div className="mb-6 text-center max-w-md z-10 px-4">
        <h2 className="text-xl font-bold text-zinc-100 tracking-tight flex items-center justify-center gap-2">
          <Palette className="text-indigo-400 w-5 h-5" /> Color Picker Redesign
        </h2>
        <p className="text-xs text-zinc-400 mt-2 leading-relaxed">
          Interactive concept of your <strong className="text-zinc-200">Option Colors</strong> view. We replaced the long row of 18 inline buttons with an elegant popover dropdown system. No clipping, beautiful tag previews, and a premium dashboard feel.
        </p>
      </div>

      {/* Field Settings Modal Replica */}
      <div className="w-full max-w-[480px] bg-[#0d0d0f] border border-white/[0.06] rounded-2xl overflow-visible shadow-[0_40px_80px_rgba(0,0,0,0.9)] z-20">
        
        {/* Modal Header */}
        <div className="px-6 pt-5 pb-3 flex items-center justify-between border-b border-white/[0.04]">
          <div className="flex items-center gap-2.5">
            <div className="p-2 rounded-xl bg-indigo-500/10 border border-indigo-500/15">
              <Sparkles size={14} className="text-indigo-400" />
            </div>
            <div>
              <span className="text-[10px] font-black uppercase text-zinc-500 tracking-widest block leading-none">Settings</span>
              <h3 className="text-[13px] font-bold text-white mt-1">Edit Field Options</h3>
            </div>
          </div>
          <button className="p-1.5 rounded-lg text-zinc-600 hover:text-white hover:bg-white/5 transition-all">
            <X size={14} />
          </button>
        </div>

        {/* Modal Tabs Simulator */}
        <div className="px-6 pt-3 flex gap-1 border-b border-white/[0.03]">
          <button 
            onClick={() => setActiveTab('colors')}
            className={`flex items-center gap-1.5 px-3 pb-2 text-[10px] font-bold uppercase tracking-wider transition-all border-b -mb-px ${
              activeTab === 'colors' 
                ? 'text-white border-indigo-400 text-indigo-400' 
                : 'text-zinc-600 border-transparent hover:text-zinc-400'
            }`}
          >
            <Palette size={12} /> Option Colors
          </button>
          <button 
            onClick={() => setActiveTab('preview')}
            className={`flex items-center gap-1.5 px-3 pb-2 text-[10px] font-bold uppercase tracking-wider transition-all border-b -mb-px ${
              activeTab === 'preview' 
                ? 'text-white border-indigo-400 text-indigo-400' 
                : 'text-zinc-600 border-transparent hover:text-zinc-400'
            }`}
          >
            <Type size={12} /> Tag Preview Demo
          </button>
        </div>

        {/* Modal Content - Scrollable Content Window */}
        <div className="p-6">
          <AnimatePresence mode="wait">
            {activeTab === 'colors' && (
              <motion.div
                initial={{ opacity: 0, y: 5 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -5 }}
                transition={{ duration: 0.15 }}
                className="space-y-4"
              >
                <div className="flex items-center justify-between">
                  <label className="text-[9px] font-black text-zinc-500 uppercase tracking-[0.2em]">
                    Manage Tags & Palette ({options.length})
                  </label>
                  <span className="text-[8px] text-zinc-600">Click color badge to edit</span>
                </div>

                {/* Main Interactive List */}
                <div className="space-y-2 max-h-[290px] overflow-y-auto pr-1 custom-scrollbar">
                  {options.map((opt, idx) => {
                    const currentColor = optionColors[opt] || '#71717a';
                    const isPopoverOpen = activePopover === opt;

                    return (
                      <div 
                        key={opt} 
                        className="group flex items-center justify-between bg-white/[0.02] hover:bg-white/[0.04] border border-white/[0.04] rounded-xl px-3.5 py-2.5 transition-all duration-200"
                      >
                        {/* Drag indicator & text label */}
                        <div className="flex items-center gap-2.5 min-w-0">
                          <GripVertical size={13} className="text-zinc-700 group-hover:text-zinc-500 cursor-grab active:cursor-grabbing shrink-0" />
                          
                          {/* Beautiful Pill Preview */}
                          <div 
                            className="px-2.5 py-1 rounded-md text-[11px] font-semibold tracking-wide truncate max-w-[180px] transition-all"
                            style={{ 
                              backgroundColor: hexToRgba(currentColor, 0.12), 
                              color: currentColor,
                              border: `1px solid ${hexToRgba(currentColor, 0.2)}`
                            }}
                          >
                            {opt}
                          </div>
                        </div>

                        {/* Redesigned Dropdown Popover Picker Trigger */}
                        <div className="relative">
                          <button
                            id={`trigger-${idx}`}
                            onClick={(e) => {
                              e.stopPropagation();
                              setActivePopover(isPopoverOpen ? null : opt);
                            }}
                            className={`flex items-center gap-2 px-2 py-1.5 rounded-lg border transition-all duration-200 ${
                              isPopoverOpen 
                                ? 'bg-indigo-500/10 border-indigo-500/30 ring-2 ring-indigo-500/20' 
                                : 'bg-black/40 hover:bg-white/[0.06] border-white/[0.06] hover:border-white/[0.12]'
                            }`}
                          >
                            {/* Colorful trigger circle */}
                            <span 
                              className="w-3.5 h-3.5 rounded-full block shadow-[0_0_8px_rgba(0,0,0,0.5)] shrink-0 transition-transform hover:scale-110" 
                              style={{ 
                                backgroundColor: currentColor,
                                boxShadow: `0 0 10px ${currentColor}35`
                              }}
                            />
                            <span className="text-[10px] font-mono text-zinc-500 font-bold group-hover:text-zinc-400">
                              {currentColor.toUpperCase()}
                            </span>
                            <ChevronDown size={11} className={`text-zinc-600 transition-transform duration-200 ${isPopoverOpen ? 'rotate-180 text-zinc-300' : ''}`} />
                          </button>

                          {/* Color Palette Popover Dropdown (SAFE FROM CLIPPING) */}
                          <AnimatePresence>
                            {isPopoverOpen && (
                              <ColorPopover 
                                currentColor={currentColor} 
                                optionName={opt}
                                onSelectColor={(color) => handleColorChange(opt, color)}
                                onClose={() => setActivePopover(null)}
                              />
                            )}
                          </AnimatePresence>
                        </div>
                      </div>
                    );
                  })}

                  {options.length === 0 && (
                    <div className="text-center py-8 bg-white/[0.01] border border-dashed border-white/[0.04] rounded-xl">
                      <p className="text-xs text-zinc-600">No options defined yet.</p>
                    </div>
                  )}
                </div>

                {/* Inline Action to add a quick option */}
                <form onSubmit={handleAddOption} className="flex gap-2 pt-1">
                  <input
                    type="text"
                    value={newOptionName}
                    onChange={(e) => setNewOptionName(e.target.value)}
                    placeholder="Create a new option..."
                    className="flex-1 bg-black/40 hover:bg-black/60 focus:bg-black/80 border border-white/[0.06] focus:border-indigo-500/40 rounded-xl px-3 py-2 text-[12px] text-zinc-200 placeholder-zinc-600 focus:outline-none transition-all font-medium"
                    maxLength={24}
                  />
                  <button 
                    type="submit"
                    className="p-2.5 rounded-xl bg-zinc-800 hover:bg-zinc-700 text-zinc-300 hover:text-white transition-all border border-white/[0.04]"
                  >
                    <Plus size={14} />
                  </button>
                </form>
              </motion.div>
            )}

            {activeTab === 'preview' && (
              <motion.div
                initial={{ opacity: 0, y: 5 }}
                animate={{ opacity: 1, y: 0 }}
                exit={{ opacity: 0, y: -5 }}
                transition={{ duration: 0.15 }}
                className="space-y-4 py-2"
              >
                <div className="space-y-1">
                  <h4 className="text-[10px] font-bold uppercase tracking-widest text-zinc-500">Live Grid Tag Layout</h4>
                  <p className="text-[11px] text-zinc-600">Here is how these tags look arranged together inside your table view or metadata card:</p>
                </div>

                <div className="bg-black/30 border border-white/[0.04] rounded-xl p-4 min-h-[140px] flex flex-wrap gap-2 items-center justify-start content-start">
                  {options.map((opt) => {
                    const clr = optionColors[opt] || '#71717a';
                    return (
                      <span 
                        key={opt}
                        className="px-3 py-1.5 rounded-lg text-xs font-semibold tracking-wide transition-all shadow-sm hover:brightness-110 cursor-default"
                        style={{ 
                          backgroundColor: hexToRgba(clr, 0.14), 
                          color: clr,
                          border: `1px solid ${hexToRgba(clr, 0.25)}`
                        }}
                      >
                        {opt}
                      </span>
                    );
                  })}
                </div>

                <div className="p-3 bg-indigo-500/[0.02] border border-indigo-500/10 rounded-xl">
                  <p className="text-[10px] text-indigo-400 leading-relaxed font-medium">
                    ✨ Pro Tip: Using soft-alpha backgrounds (`rgba`) mapped from your hex codes elevates the table visual rhythm and matches best-in-class tools like Slack, Notion, and Linear!
                  </p>
                </div>
              </motion.div>
            )}
          </AnimatePresence>
        </div>

        {/* Modal Footer */}
        <div className="px-6 py-4 border-t border-white/[0.04] flex items-center justify-between">
          <button 
            onClick={handleReset}
            className="px-3.5 py-2 rounded-xl bg-white/[0.03] hover:bg-white/[0.06] border border-white/[0.06] text-zinc-500 hover:text-zinc-300 transition-all text-[10px] font-bold uppercase tracking-wider"
          >
            <RotateCcw size={11} className="inline mr-1" /> Reset Defaults
          </button>
          
          <button 
            onClick={handleSave}
            className={`px-5 py-2.5 rounded-xl text-[10px] font-black uppercase tracking-wider transition-all active:scale-95 ${
              saved 
                ? 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30 shadow-[0_0_15px_rgba(16,185,129,0.1)]' 
                : 'bg-white text-black hover:bg-zinc-200'
            }`}
          >
            {saved ? <><Check size={11} className="inline mr-1" /> Saved Changes</> : 'Save Options'}
          </button>
        </div>
      </div>

      {/* User Guide and Technical Note */}
      <div className="mt-6 text-zinc-600 text-[10px] max-w-sm text-center">
        To dismiss a popover, click anywhere outside or pick a color. Floating popovers use local bounding boxes to align safely.
      </div>

    </div>
  );
}

// Reusable Popover component built for non-clipping operations
function ColorPopover({ currentColor, optionName, onSelectColor, onClose }) {
  const popoverRef = useRef(null);

  // Close when user clicks outside the popover
  useEffect(() => {
    function handleClickOutside(event) {
      if (popoverRef.current && !popoverRef.current.contains(event.target)) {
        onClose();
      }
    }
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [onClose]);

  return (
    <motion.div
      ref={popoverRef}
      initial={{ opacity: 0, scale: 0.95, y: -4 }}
      animate={{ opacity: 1, scale: 1, y: 0 }}
      exit={{ opacity: 0, scale: 0.95, y: -4 }}
      transition={{ duration: 0.12, ease: "easeOut" }}
      className="absolute right-0 mt-2 w-[220px] bg-[#121215] border border-white/[0.08] rounded-xl p-3 shadow-[0_12px_30px_rgba(0,0,0,0.85)] z-50 origin-top-right"
    >
      {/* Popover Header */}
      <div className="flex items-center justify-between pb-2 mb-2.5 border-b border-white/[0.04]">
        <div className="flex items-center gap-1.5">
          <Hash size={10} className="text-zinc-500" />
          <span className="text-[9px] font-black uppercase text-zinc-400 tracking-wider">Select Color</span>
        </div>
        <span 
          className="text-[9px] font-mono font-bold uppercase"
          style={{ color: currentColor }}
        >
          {currentColor}
        </span>
      </div>

      {/* Color Grid ($6 \times 3$) */}
      <div className="grid grid-cols-6 gap-1.5">
        {OPTION_COLORS.map((color) => {
          const isSelected = currentColor === color;
          return (
            <button
              key={color}
              type="button"
              onClick={() => {
                onSelectColor(color);
                onClose(); // Auto-close on select for a smooth UX
              }}
              className="group relative w-6 h-6 rounded-full transition-transform active:scale-90 flex items-center justify-center focus:outline-none"
              style={{ backgroundColor: color }}
              title={color}
            >
              {/* Selective internal tick or hover ring */}
              {isSelected ? (
                <div className="w-2 h-2 rounded-full bg-white shadow-sm" />
              ) : (
                <div className="w-0 h-0 group-hover:w-1.5 group-hover:h-1.5 rounded-full bg-white/40 transition-all duration-150" />
              )}
            </button>
          );
        })}
      </div>

      {/* Quick custom indicator to showcase details */}
      <div className="mt-2.5 pt-2 border-t border-white/[0.03] flex items-center justify-between">
        <span className="text-[8px] text-zinc-600 font-medium">Standard Palette</span>
        <div 
          className="w-2 h-2 rounded-full animate-pulse" 
          style={{ backgroundColor: currentColor }} 
        />
      </div>
    </motion.div>
  );
}
