  50% { opacity: 0.35; transform: scale(1.03); }

  from { opacity: 0; transform: translateX(20px); filter: blur(4px); }
  to { opacity: 1; transform: translateX(0); filter: blur(0); }

}

export default function App() {
  const [currentSlide, setCurrentSlide] = useState(0);
  const [direction, setDirection] = useState('next');
  const [isTransitioning, setIsTransitioning] = useState(false);
  const [studioLightIntensity, setStudioLightIntensity] = useState(40); // 0 to 100 scale
  const [isAutoplay, setIsAutoplay] = useState(false);
  
  const TOTAL_SLIDES = 3;

  useEffect(() => {
    if (!isAutoplay) return;
    const interval = setInterval(() => {
      handleNext();
    }, 7000);
    return () => clearInterval(interval);
  }, [isAutoplay, currentSlide]);

  const handleNext = () => {
    if (isTransitioning) return;
    setIsTransitioning(true);
    setDirection('next');
    setTimeout(() => {
      setCurrentSlide((prev) => (prev + 1) % TOTAL_SLIDES);
      setIsTransitioning(false);
    }, 400);
  };

  const handlePrev = () => {
    if (isTransitioning) return;
    setIsTransitioning(true);
    setDirection('prev');
    setTimeout(() => {
      setCurrentSlide((prev) => (prev - 1 + TOTAL_SLIDES) % TOTAL_SLIDES);
      setIsTransitioning(false);
    }, 400);
  };

  useEffect(() => {
    const handleKeyDown = (e) => {
      if (e.key === 'ArrowRight' || e.key === 'Space') {
        e.preventDefault();
        handleNext();
      } else if (e.key === 'ArrowLeft') {
        e.preventDefault();
        handlePrev();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [currentSlide, isTransitioning]);

  return (
    <div className="w-full min-h-screen bg-[#000000] text-white font-sans antialiased overflow-x-hidden relative flex flex-col justify-between custom-easing duration-700">
      <style>{CSS_ANIMATIONS}</style>
      
      {/* Monochromatic Studio Spotlight Background - STRICTLY BLACK & WHITE FOCUS */}
      <div className="absolute inset-0 overflow-hidden pointer-events-none z-0">
        <div 
          style={{
            opacity: studioLightIntensity / 100 * 0.25,
            background: 'radial-gradient(circle at 50% 40%, rgba(255, 255, 255, 0.4) 0%, rgba(0, 0, 0, 0) 70%)',
            transition: 'opacity 0.5s ease-out'
          }}
          className="absolute inset-0 w-full h-full animate-pulse-glow" 
        />
        <div className="absolute top-0 left-0 w-full h-full bg-[radial-gradient(ellipse_at_top,_var(--tw-gradient-stops))] from-zinc-900/40 via-black to-black opacity-60" />
      </div>

      {/* Presentation Header */}
      <header className="relative z-50 px-8 py-6 flex items-center justify-between border-b border-white/5 max-w-7xl mx-auto w-full">
        <div className="flex items-center space-x-3 group cursor-pointer">
          <div className="w-9 h-9 rounded-lg bg-white text-black flex items-center justify-center shadow-lg transition-transform duration-300 group-hover:scale-105">
            <span className="font-bold text-base tracking-tight">C</span>
          </div>
          <div>
            <span className="font-medium text-base tracking-tight text-zinc-100">Chronos</span>
            <span className="text-[10px] block text-zinc-500 tracking-wider uppercase font-semibold">Atelier Edition</span>
          </div>
        </div>

        {/* Studio Lighting Controls */}
        <div className="flex items-center space-x-6 bg-zinc-900/60 border border-white/5 px-4 py-2 rounded-full">
          <div className="flex items-center space-x-2">
            <Sliders size={12} className="text-zinc-400" />
            <span className="text-[10px] font-semibold text-zinc-400 uppercase tracking-wider">Studio Light</span>
          </div>
          <input 
            type="range" 
            min="10" 
            max="100" 
            value={studioLightIntensity}
            onChange={(e) => setStudioLightIntensity(Number(e.target.value))}
            className="w-20 h-1 bg-zinc-800 rounded-lg appearance-none cursor-pointer accent-white"
            title="Adjust cinematic spotlight background brightness"
          />
          <button 
            onClick={() => setIsAutoplay(!isAutoplay)}
            className={`px-3 py-1 rounded-full text-[10px] font-bold tracking-widest transition-all ${isAutoplay ? 'bg-white text-black' : 'border border-zinc-800 text-zinc-500 hover:text-zinc-300'}`}
          >
            {isAutoplay ? 'AUTOPLAY' : 'MANUAL'}
          </button>
        </div>
      </header>

      {/* Main Grid: Anchored Text Left, Vibrant Interactive Canvas Right */}
      <main className="flex-1 relative z-10 flex items-center justify-center px-4 md:px-12 py-12 max-w-7xl mx-auto w-full select-none">
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-12 lg:gap-16 items-center w-full">
          
          {/* Static Left Side (Anchored Value Proposition for Luxury Salon) */}
          <div className="lg:col-span-5 flex flex-col justify-center space-y-6 text-left relative z-20">
            <div className="inline-flex items-center space-x-2 px-3 py-1.5 rounded-full border border-white/10 bg-white/5 text-zinc-300 text-xs font-semibold uppercase tracking-wider w-fit">
              <Sparkles size={12} className="text-cyan-400" />
              <span>The Atelier Experience</span
            </div>
            
            <h1 className="text-5xl md:text-7xl lg:text-8xl font-black tracking-tight leading-[0.9] text-transparent bg-clip-text bg-gradient-to-b from-white via-zinc-200 to-zinc-600">
              Sculpting<br/>
              Time.
            </h1>
            
            <p className="text-lg text-zinc-400 font-light max-w-md leading-relaxed">
              Chronos is an elite scheduling canvas engineered for luxury salons. Elevate client hospitality with seamless booking, visual station allocation, and a flawless guest journey.
            </p>

            <div className="pt-4 flex flex-wrap items-center gap-4">
              <span className="flex items-center space-x-2 text-[10px] font-bold text-zinc-400 tracking-widest uppercase">
                <span className="w-1.5 h-1.5 rounded-full bg-cyan-400 shadow-[0_0_8px_#06b6d4]" />
                <span>Guest Journey</span>
              </span>
              <span className="flex items-center space-x-2 text-[10px] font-bold text-zinc-400 tracking-widest uppercase">
                <span className="w-1.5 h-1.5 rounded-full bg-violet-400 shadow-[0_0_8px_#8b5cf6]" />
                <span>Station Allocation</span>
              </span>
              <span className="flex items-center space-x-2 text-[10px] font-bold text-zinc-400 tracking-widest uppercase">
                <span className="w-1.5 h-1.5 rounded-full bg-rose-400 shadow-[0_0_8px_#f43f5e]" />
                <span>Stylist Harmony</span>
              </span>
            </div>

            {/* Micro Indicator highlighting current view scope */}
            <div className="pt-8 border-t border-white/5 mt-6 hidden lg:block">
              <span className="text-[10px] text-zinc-500 font-bold uppercase tracking-widest block mb-2">Active Spec Scope</span>
              <div className="flex space-x-6 text-xs">
                <span className={`transition-all duration-300 ${currentSlide === 0 ? 'text-cyan-400 font-semibold' : 'text-zinc-600'}`}>01. Guest Card</span>
                <span className={`transition-all duration-300 ${currentSlide === 1 ? 'text-violet-400 font-semibold' : 'text-zinc-600'}`}>02. Floor Matrix</span>
                <span className={`transition-all duration-300 ${currentSlide === 2 ? 'text-rose-400 font-semibold' : 'text-zinc-600'}`}>03. Request Parser</span>
              </div>
            </div>
          </div>

          {/* Dynamic Active Right Column */}
          <div className="lg:col-span-7 flex items-center justify-center relative min-h-[520px] w-full">
            <div 
              className={`w-full transition-all duration-400 transform custom-easing ${
                isTransitioning 
                  ? (direction === 'next' ? 'opacity-0 translate-x-4 scale-98 filter blur-sm' : 'opacity-0 translate-x--4 scale-98 filter blur-sm') 
                  : 'opacity-100 translate-x-0 scale-100 filter blur-none'
              }`}
            >
              {currentSlide === 0 && <RightHeroWidget />}
              {currentSlide === 1 && <RightCalendarGrid />}
              {currentSlide === 2 && <RightNlpSandbox />}
            </div>
          </div>

        </div>
      </main>

      {/* Presentation Footer */}
      <footer className="relative z-50 px-8 py-8 max-w-7xl mx-auto w-full flex flex-col md:flex-row items-center justify-between gap-6 border-t border-white/5">
        {/* Navigation Indicator Dots with slide color themes */}
        <div className="flex items-center space-x-3">
          {[0, 1, 2].map((idx) => {
            const titles = ["01/ Guest Schedule", "02/ Atelier Floor Matrix", "03/ Guest Request Parser"];
            const activeColors = [
              'bg-gradient-to-r from-cyan-400 to-blue-500 shadow-[0_0_15px_rgba(6,182,212,0.5)]',
              'bg-gradient-to-r from-violet-500 to-fuchsia-500 shadow-[0_0_15px_rgba(139,92,246,0.5)]',
              'bg-gradient-to-r from-rose-500 to-amber-500 shadow-[0_0_15px_rgba(244,63,94,0.5)]'
            ];
            const textActiveColors = ['text-cyan-300', 'text-violet-300', 'text-rose-300'];
            
            return (
              <button
                key={idx}
                onClick={() => {
                  if (isTransitioning) return;
                  setIsTransitioning(true);
                  setDirection(idx > currentSlide ? 'next' : 'prev');
                  setTimeout(() => {
                    setCurrentSlide(idx);
                    setIsTransitioning(false);
                  }, 400);
                }}
                className="group relative py-2"
              >
                <div className={`h-1.5 rounded-full transition-all duration-500 ${currentSlide === idx ? `w-20 ${activeColors[idx]}` : 'w-8 bg-zinc-800 hover:bg-zinc-600'}`} />
                <span className={`absolute top-6 left-0 text-[9px] tracking-wider font-semibold uppercase whitespace-nowrap transition-all duration-300 ${currentSlide === idx ? `${textActiveColors[idx]} opacity-100 translate-y-0` : 'text-zinc-600 opacity-0 -translate-y-1'}`}>
                  {titles[idx]}
                </span>
              </button>
            );
          })}
        </div>

        {/* Navigation Control Buttons */}
        <div className="flex items-center space-x-3">
          <button
            onClick={handlePrev}
            className="w-12 h-12 rounded-full border border-zinc-800 bg-zinc-950/40 text-white flex items-center justify-center hover:bg-zinc-900 hover:border-zinc-700 transition-all duration-300 hover:scale-105 active:scale-95"
            aria-label="Previous slide"
          >
            <ChevronLeft size={18} />
          </button>
          <button
            onClick={handleNext}
            className="w-28 h-12 rounded-full bg-white text-black flex items-center justify-center gap-2 font-semibold transition-all duration-300 hover:scale-105 active:scale-95 hover:bg-zinc-200"
          >
            <span>Advance</span>
            <ArrowRight size={14} />
          </button>
        </div>
      </footer>
    </div>
  );
}

                  <span className={`text-[9px] font-bold uppercase tracking-wider px-2 py-0.5 rounded border border-white/5 bg-white/5 ${ev.colorText}`}>
                    {ev.category}
                  </span>
                  <span className="text-[10px] font-mono text-zinc-400">{ev.time}</span>
                </div>
              </div>
            ))}
          </div>
        ) : (
          <p className="text-xs text-zinc-500 italic">No large-scale bookings scheduled. Open stations available for boutique appointments.</p>
        )}
      </div>

    </div>
  );
}
