import React, { useState, useEffect, useRef } from 'react';

// ==========================================
// CHARACTER REEL DATA
// ==========================================
const DIAL_REELS = [
  ["05", "GMT", "12:00", "MON", "OCT", "22", "UTC", "09:45", "WED", "DEC", "14", "EST", "18:30", "PST", "B"],
  ["12", "CET", "15:30", "TUE", "JAN", "08", "JST", "21:15", "THU", "APR", "29", "AST", "11:00", "MST", "O"],
  ["19", "PST", "08:15", "WED", "FEB", "31", "MST", "14:30", "FRI", "MAY", "03", "CST", "23:45", "AEST", "O"],
  ["27", "EST", "23:00", "THU", "MAR", "11", "NST", "07:00", "SAT", "JUN", "18", "HST", "16:15", "AKST", "K"],
  ["03", "JST", "06:45", "FRI", "APR", "15", "AEST", "19:30", "SUN", "JUL", "25", "SST", "08:00", "ChST", "I"],
  ["14", "AEST", "17:00", "SAT", "MAY", "29", "ChST", "11:15", "MON", "AUG", "07", "GMT", "20:30", "WET", "N"],
  ["22", "NZST", "10:30", "SUN", "JUN", "04", "WET", "16:00", "TUE", "SEP", "13", "CET", "05:15", "EET", "G"]
];

const SYMMETRIC_WAVE = {
  easing: 'cubic-bezier(0.22, 1, 0.36, 1)',
  delays: [0, 200, 400, 500, 400, 200, 0],
  durations: [1500, 1500, 1500, 1500, 1500, 1500, 1500]
};

export default function App() {
  const [playState, setPlayState] = useState('playing');

  const handleTrigger = () => {
    setPlayState('resetting');
    setTimeout(() => {
      setPlayState('playing');
    }, 50);
  };

  useEffect(() => {
    handleTrigger();
  }, []);

  return (
    <div 
      className="min-h-screen bg-black text-white flex flex-col items-center justify-center select-none cursor-pointer overflow-hidden relative"
      onClick={handleTrigger}
    >
      {}
      <div className="absolute inset-x-0 h-[1px] bg-zinc-900/40 top-[calc(50%-1.1em)] sm:top-[calc(50%-1.6em)] pointer-events-none" />
      <div className="absolute inset-x-0 h-[1px] bg-zinc-900/40 top-[calc(50%+1.1em)] sm:top-[calc(50%+1.6em)] pointer-events-none" />

      {}
      <div className="z-10 flex items-center justify-center w-full">
        <h2 
          style={{
            fontFamily: '-apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Helvetica, Arial, sans-serif',
          }}
          className="flex font-black leading-[0.98] tracking-[-0.05em] text-5xl md:text-7xl lg:text-[4rem]"
        >
          {DIAL_REELS.map((reel, dialIdx) => {
            const delayVal = SYMMETRIC_WAVE.delays[dialIdx];
            const durationVal = SYMMETRIC_WAVE.durations[dialIdx];
            const translateY = playState === 'playing' ? (14 * -0.98) : 0;

            return (
              <span 
                key={dialIdx}
                style={{ height: '0.98em' }}
                // Tightened width for "almost touching" effect
                className="inline-block overflow-hidden relative border-r border-zinc-950 last:border-0 w-[0.85em]"
              >
                <span 
                  className="flex flex-col items-center"
                  style={{
                    transform: `translateY(${translateY}em)`,
                    transitionProperty: playState === 'resetting' ? 'none' : 'transform',
                    transitionDuration: `${durationVal}ms`,
                    transitionDelay: `${delayVal}ms`,
                    transitionTimingFunction: SYMMETRIC_WAVE.easing
                  }}
                >
                  {reel.map((char, charIdx) => {
                    const isTarget = charIdx === 14;
                    return (
                      <span 
                        key={charIdx} 
                        style={{ height: '0.98em' }}
                        className={`flex items-center justify-center w-full text-center ${
                          isTarget 
                            ? 'bg-clip-text text-transparent bg-gradient-to-b from-white via-zinc-100 to-zinc-400 font-black' 
                            // Lighter, more legible scroll numbers
                            : 'text-zinc-400 font-medium scale-90 opacity-70 filter blur-[0.2px]'
                        }`}
                      >
                        {char}
                      </span>
                    );
                  })}
                </span>
              </span>
            );
          })}
        </h2>
      </div>

      <div className="absolute bottom-8 text-[10px] tracking-[0.2em] uppercase font-mono text-zinc-600 animate-pulse pointer-events-none">
        Click anywhere to replay
      </div>
    </div>
  );
}
