create table public.appointments (
  id uuid not null default gen_random_uuid (),
  lead_id bigint null,
  client_name text not null,
  date date not null,
  time text not null,
  duration integer null default 30,
  status text null default 'pending'::text,
  assigned_receptionist text null,
  notes text null,
  scenario_id uuid null,
  created_at timestamp with time zone null default now(),
  user_id uuid null,
  person_id bigint null,
  constraint appointments_pkey primary key (id),
  constraint appointments_lead_id_fkey foreign KEY (lead_id) references people (id) on delete CASCADE,
  constraint appointments_user_id_fkey foreign KEY (user_id) references users (id) on delete CASCADE,
  constraint appointments_status_check check (
    (
      status = any (
        array[
          'pending'::text,
          'confirmed'::text,
          'cancelled'::text,
          'completed'::text,
          'missed'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_appointments_date on public.appointments using btree (date) TABLESPACE pg_default;

create index IF not exists idx_appointments_status on public.appointments using btree (status) TABLESPACE pg_default;

create index IF not exists idx_appointments_lead on public.appointments using btree (lead_id) TABLESPACE pg_default;

create index IF not exists idx_appointments_user_id on public.appointments using btree (user_id) TABLESPACE pg_default;






create table public.businesses (
  id bigint generated by default as identity not null,
  owner_id uuid null,
  name text not null,
  phone text null,
  email text null,
  address text null,
  city text null,
  state text null,
  zip text null,
  website text null,
  about_us text null default ''::text,
  policies text null default ''::text,
  faq text null default ''::text,
  business_hours text null default ''::text,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  user_id uuid not null,
  constraint businesses_pkey primary key (id),
  constraint businesses_user_id_fkey foreign KEY (user_id) references users (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_businesses_user_id on public.businesses using btree (user_id) TABLESPACE pg_default;

  last_outbound_call_at timestamp with time zone null,
  last_call_status text null,
  last_intent text null,

create table public.scenarios (
  id uuid not null default gen_random_uuid (),
  name text not null,
  description text null,
  nodes_data jsonb not null default '[]'::jsonb,
  edges_data jsonb not null default '[]'::jsonb,
  status text not null default 'draft'::text,
  created_by uuid null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  assigned_to text null,
  notes text null,
  is_active boolean null default true,
  schedule_config jsonb null default '{}'::jsonb,
  last_fired_at timestamp with time zone null,
  constraint scenarios_pkey primary key (id),
  constraint scenarios_created_by_fkey foreign KEY (created_by) references auth.users (id),
  constraint scenarios_status_check check (
    (
      status = any (
        array['draft'::text, 'active'::text, 'archived'::text]
      )
    )
  )
) TABLESPACE pg_default;
