create table public.users (
  id uuid not null default gen_random_uuid (),
  email text not null,
  full_name text null,
  created_at timestamp with time zone null default now(),
  constraint users_pkey primary key (id),
  constraint users_email_key unique (email)
) TABLESPACE pg_default;

create table public.services (
  id uuid not null,
  name text not null,
  description text null,
  price_type text null default 'fixed'::text,
  price_min numeric null,
  price_max numeric null,
  unit text null default ''::text,
  category text null default 'General'::text,
  is_active boolean null default true,
  sort_order integer null default 0,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  user_id uuid not null,
  constraint services_pkey primary key (id),
  constraint services_user_id_fkey foreign KEY (user_id) references users (id) on delete CASCADE,
  constraint services_price_type_check check (
    (
      price_type = any (
        array[
          'fixed'::text,
          'starting_at'::text,
          'range'::text,
          'quote'::text,
          'free'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_services_active on public.services using btree (is_active) TABLESPACE pg_default;

create index IF not exists idx_services_category on public.services using btree (category) TABLESPACE pg_default;

create index IF not exists idx_services_user_id on public.services using btree (user_id) TABLESPACE pg_default;



create table public.scenarios (
  id uuid not null default gen_random_uuid (),
  name text not null,
  description text null,
  nodes_data jsonb not null default '[]'::jsonb,
  edges_data jsonb not null default '[]'::jsonb,
  status text not null default 'draft'::text,
  created_by uuid null,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  assigned_to text null,
  notes text null,
  is_active boolean null default true,
  schedule_config jsonb null default '{}'::jsonb,
  last_fired_at timestamp with time zone null,
  user_id uuid null,
  constraint scenarios_pkey primary key (id),
  constraint scenarios_created_by_fkey foreign KEY (created_by) references auth.users (id),
  constraint scenarios_status_check check (
    (
      status = any (
        array['draft'::text, 'active'::text, 'archived'::text]
      )
    )
  )
) TABLESPACE pg_default;

          'no_answer'::text,
          'missed'::text

    )
  )
) TABLESPACE pg_default;

create table public.appointments (
  id uuid not null default gen_random_uuid (),
  lead_id bigint null,
  client_name text not null,
  date date not null,
  time text not null,
  duration integer null default 30,
  status text null default 'pending'::text,
  assigned_receptionist text null,
  notes text null,
  scenario_id uuid null,
  created_at timestamp with time zone null default now(),
  user_id uuid null,
  people_id bigint null,
  constraint appointments_pkey primary key (id),
  constraint appointments_lead_id_fkey foreign KEY (lead_id) references people (id) on delete CASCADE,
  constraint appointments_user_id_fkey foreign KEY (user_id) references users (id) on delete CASCADE,
  constraint appointments_status_check check (
    (
      status = any (
        array[
          'pending'::text,
          'confirmed'::text,
          'cancelled'::text
        ]
      )
    )
  )
) TABLESPACE pg_default;

create index IF not exists idx_appointments_status on public.appointments using btree (status) TABLESPACE pg_default;
