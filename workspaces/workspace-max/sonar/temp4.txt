import React, { useState, useEffect } from 'react';
import { 
  ChevronRight, 
  ChevronLeft, 
  Sparkles, 
  Check, 
  Loader2, 
  Globe, 
  Users, 
  Target,
  ArrowRight,
  ShieldCheck,
  Building
} from 'lucide-react';

// Inject custom keyframes for ultra-smooth Apple-like transitions tailored for dark mode
const styleSheet = document.createElement("style");
styleSheet.innerText = `
  @keyframes fadeIn {
    from { opacity: 0; transform: translateY(8px); }
    to { opacity: 1; transform: translateY(0); }
  }
  @keyframes fadeOut {
    from { opacity: 1; transform: translateY(0); }
    to { opacity: 0; transform: translateY(-8px); }
  }
  @keyframes pulseSubtle {
    0%, 100% { opacity: 1; }
    50% { opacity: 0.4; }
  }
  .animate-fade-in {
    animation: fadeIn 0.6s cubic-bezier(0.16, 1, 0.3, 1) forwards;
  }
  .animate-pulse-subtle {
    animation: pulseSubtle 2.2s infinite ease-in-out;
  }
  input:focus {
    box-shadow: 0 0 0 4px rgba(255, 255, 255, 0.08);
  }
  .dark-focus:focus {
    box-shadow: 0 0 0 4px rgba(255, 255, 255, 0.15);
  }
`;
document.head.appendChild(styleSheet);

export default function App() {
  const [step, setStep] = useState(1);
  const [direction, setDirection] = useState('next'); // 'next' | 'prev'
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [onboardingComplete, setOnboardingComplete] = useState(false);
  const [shake, setShake] = useState(false);

  // Form State
  const [formData, setFormData] = useState({
    businessName: '',
    workspaceUrl: '',
    teamSize: '',
    primaryGoal: '',
  });

  // Errors for inline feedback
  const [errors, setErrors] = useState({
    businessName: '',
    workspaceUrl: '',
    teamSize: '',
    primaryGoal: '',
  });

  const validateStep = (currentStep) => {
    let isValid = true;
    const newErrors = { ...errors };

    if (currentStep === 1) {
      if (!formData.businessName.trim()) {
        newErrors.businessName = 'Please enter your business name to continue.';
        isValid = false;
      } else {
        newErrors.businessName = '';
      }
    }

    if (currentStep === 2) {
      const urlPattern = /^[a-z0-9-]+$/;
      if (!formData.workspaceUrl.trim()) {
        newErrors.workspaceUrl = 'Please choose a workspace URL.';
        isValid = false;
      } else if (!urlPattern.test(formData.workspaceUrl)) {
        newErrors.workspaceUrl = 'Only lowercase letters, numbers, and hyphens allowed.';
        isValid = false;
      } else {
        newErrors.workspaceUrl = '';
      }
    }

    if (currentStep === 3) {
      if (!formData.teamSize) {
        newErrors.teamSize = 'Please select your team scale.';
        isValid = false;
      } else {
        newErrors.teamSize = '';
      }
    }

    if (currentStep === 4) {
      if (!formData.primaryGoal) {
        newErrors.primaryGoal = 'Please select your primary objective.';
        isValid = false;
      } else {
        newErrors.primaryGoal = '';
      }
    }

    setErrors(newErrors);
    
    if (!isValid) {
      setShake(true);
      setTimeout(() => setShake(false), 500);
    }
    
    return isValid;
  };

  const handleNext = () => {
    if (validateStep(step)) {
      setDirection('next');
      if (step < 4) {
        setStep(prev => prev + 1);
      } else {
        triggerWorkspaceCreation();
      }
    }
  };

  const handleBack = () => {
    if (step > 1) {
      setDirection('prev');
      setStep(prev => prev - 1);
    }
  };

  const triggerWorkspaceCreation = () => {
    setIsSubmitting(true);
    // Simulate Apple-level fluid processing transition in dark mode
    setTimeout(() => {
      setIsSubmitting(false);
      setOnboardingComplete(true);
    }, 2800);
  };

  const handleKeyDown = (e) => {
    if (e.key === 'Enter') {
      e.preventDefault();
      handleNext();
    }
  };

  const updateField = (field, value) => {
    setFormData(prev => ({ ...prev, [field]: value }));
    // Generate auto-workspace URL based on business name
    if (field === 'businessName') {
      const autoUrl = value
        .toLowerCase()
        .replace(/[^a-z0-9]/g, '-')
        .replace(/-+/g, '-')
        .replace(/^-|-$/g, '');
      setFormData(prev => ({ 
        ...prev, 
        businessName: value,
        workspaceUrl: autoUrl 
      }));
    }
    // Instantly clear error on change
    if (errors[field]) {
      setErrors(prev => ({ ...prev, [field]: '' }));
    }
  };

  const totalSteps = 4;
  const progressPercent = (step / totalSteps) * 100;

  return (
    <div className="min-h-screen bg-black text-neutral-100 font-sans flex flex-col justify-between selection:bg-neutral-800 antialiased">
      
      {}
      {/* Header Container */}
      <header className="w-full max-w-lg mx-auto px-6 pt-8 md:pt-12 flex justify-between items-center h-16">
        <div className="flex items-center space-x-2">
          <div className="w-8 h-8 rounded-lg bg-neutral-900 border border-neutral-800 flex items-center justify-center text-white font-semibold shadow-sm">
            <Sparkles className="w-4 h-4 text-neutral-200" />
          </div>
          <span className="text-sm font-medium tracking-tight text-neutral-200">Aura Workspace</span>
        </div>

        {/* Dynamic Navigation Step Display */}
        {!onboardingComplete && !isSubmitting && (
          <div className="text-xs font-mono text-neutral-500">
            {step} of {totalSteps}
          </div>
        )}
      </header>

      {/* Main Form Area */}
      <main className="flex-1 flex flex-col justify-center px-6 py-12 md:py-24">
        <div className={`w-full max-w-md mx-auto transition-transform duration-300 ${shake ? 'animate-bounce' : ''}`}>
          
          {/* Progress Indicator Line */}
          {!onboardingComplete && !isSubmitting && (
            <div className="w-full bg-neutral-900 h-[2px] rounded-full mb-12 overflow-hidden">
              <div 
                className="bg-neutral-200 h-full transition-all duration-500 ease-out shadow-[0_0_8px_rgba(255,255,255,0.5)]"
                style={{ width: `${progressPercent}%` }}
              />
            </div>
          )}

          {}
          {isSubmitting ? (
            <div className="text-center py-12 animate-fade-in flex flex-col items-center">
              <Loader2 className="w-8 h-8 animate-spin text-neutral-200 mb-6" />
              <h2 className="text-xl font-medium tracking-tight text-neutral-100 mb-2">Creating your workspace</h2>
              <p className="text-sm text-neutral-400 max-w-xs animate-pulse-subtle leading-relaxed">
                Securing custom domain, configuring parameters, and spinning up database clusters...
              </p>
            </div>
          ) : onboardingComplete ? (
            <div className="animate-fade-in space-y-8 py-4">
              <div className="w-12 h-12 rounded-full bg-neutral-900 border border-emerald-900/50 flex items-center justify-center text-emerald-400 mb-6">
                <Check className="w-6 h-6" />
              </div>
              
              <div className="space-y-3">
                <h1 className="text-3xl font-semibold tracking-tight text-neutral-50">
                  Welcome to {formData.businessName || "your new space"}.
                </h1>
                <p className="text-base text-neutral-400 leading-relaxed">
                  Your customized Aura workspace is ready. We have configured the environment for a team size of <span className="text-neutral-200 font-medium">{formData.teamSize || "1"}</span>, tailored specifically to help you <span className="text-neutral-200 font-medium">{formData.primaryGoal?.toLowerCase() || "achieve your goals"}</span>.
                </p>
              </div>

              {/* Workspace Preview Artifact */}
              <div className="p-6 bg-neutral-900/60 rounded-2xl border border-neutral-800 shadow-xl space-y-4">
                <div className="flex items-center justify-between border-b border-neutral-800/60 pb-4">
                  <div className="flex items-center space-x-3">
                    <div className="w-10 h-10 rounded-xl bg-white flex items-center justify-center text-neutral-950 font-semibold text-lg">
                      {formData.businessName.charAt(0).toUpperCase()}
                    </div>
                    <div>
                      <div className="text-sm font-semibold text-neutral-100">{formData.businessName}</div>
                      <div className="text-xs text-neutral-500">{formData.workspaceUrl}.aura.co</div>
                    </div>
                  </div>
                  <span className="text-xs bg-emerald-950/40 text-emerald-400 px-2.5 py-1 rounded-full border border-emerald-900/40 font-medium">Active</span>
                </div>
                
                <div className="grid grid-cols-2 gap-4 text-xs">
                  <div>
                    <span className="text-neutral-500 block mb-1">Configuration</span>
                    <span className="text-neutral-300 font-medium">Business Suite</span>
                  </div>
                  <div>
                    <span className="text-neutral-500 block mb-1">Integration</span>
                    <span className="text-neutral-300 font-medium">Standard SSO</span>
                  </div>
                </div>
              </div>

              <div className="pt-4">
                <button 
                  onClick={() => {
                    setOnboardingComplete(false);
                    setStep(1);
                    setFormData({ businessName: '', workspaceUrl: '', teamSize: '', primaryGoal: '' });
                  }}
                  className="w-full py-3.5 bg-white text-neutral-950 rounded-xl font-medium text-sm transition-all hover:bg-neutral-200 active:scale-[0.98] focus:outline-none focus:ring-2 focus:ring-white focus:ring-offset-2 focus:ring-offset-black flex items-center justify-center space-x-2"
                >
                  <span>Launch Workspace</span>
                  <ArrowRight className="w-4 h-4" />
                </button>
              </div>
            </div>
          ) : (
            <div className="space-y-8 animate-fade-in">
              
              {}
              {/* Step 1: Company Name */}
              {step === 1 && (
                <div className="space-y-6">
                  <div className="space-y-3">
                    <h1 className="text-2xl md:text-3xl font-semibold tracking-tight text-neutral-100 animate-fade-in">
                      What is your business called?
                    </h1>
                    <p className="text-sm md:text-base text-neutral-400 leading-relaxed">
                      We'll use this to set up your shared brand resources, communications, and workspaces.
                    </p>
                  </div>
                  
                  <div className="space-y-2">
                    <input
                      type="text"
                      value={formData.businessName}
                      onChange={(e) => updateField('businessName', e.target.value)}
                      onKeyDown={handleKeyDown}
                      placeholder="e.g., Acme Corporation"
                      autoFocus
                      className="w-full px-5 py-4 bg-neutral-900/50 border border-neutral-800 rounded-xl text-neutral-100 placeholder:text-neutral-600 text-base focus:outline-none focus:border-neutral-700 transition-all"
                    />
                    {errors.businessName && (
                      <p className="text-xs text-rose-500 ml-1">{errors.businessName}</p>
                    )}
                  </div>
                </div>
              )}

              {}
              {/* Step 2: Custom URL */}
              {step === 2 && (
                <div className="space-y-6">
                  <div className="space-y-3">
                    <h1 className="text-2xl md:text-3xl font-semibold tracking-tight text-neutral-100 animate-fade-in">
                      Claim your workspace domain.
                    </h1>
                    <p className="text-sm md:text-base text-neutral-400 leading-relaxed">
                      Your team will use this unique URL address to securely access your main digital dashboard.
                    </p>
                  </div>

                  <div className="space-y-2">
                    <div className="relative flex items-center">
                      <input
                        type="text"
                        value={formData.workspaceUrl}
                        onChange={(e) => updateField('workspaceUrl', e.target.value)}
                        onKeyDown={handleKeyDown}
                        placeholder="acme"
                        autoFocus
                        className="w-full pl-5 pr-24 py-4 bg-neutral-900/50 border border-neutral-800 rounded-xl text-neutral-100 placeholder:text-neutral-600 text-base focus:outline-none focus:border-neutral-700 transition-all font-mono text-sm"
                      />
                      <span className="absolute right-5 text-sm font-medium text-neutral-500 font-mono pointer-events-none">
                        .aura.co
                      </span>
                    </div>
                    {errors.workspaceUrl ? (
                      <p className="text-xs text-rose-500 ml-1">{errors.workspaceUrl}</p>
                    ) : (
                      <p className="text-xs text-neutral-500 ml-1">
                        Use lowercase letters, numbers, and hyphens only.
                      </p>
                    )}
                  </div>
                </div>
              )}

              {}
              {/* Step 3: Team Size Select */}
              {step === 3 && (
                <div className="space-y-6">
                  <div className="space-y-3">
                    <h1 className="text-2xl md:text-3xl font-semibold tracking-tight text-neutral-100 animate-fade-in">
                      How large is your team?
                    </h1>
                    <p className="text-sm md:text-base text-neutral-400 leading-relaxed">
                      We'll automatically configure optimization defaults based on your current workforce size.
                    </p>
                  </div>

                  <div className="space-y-3">
                    {[
                      { label: 'Just myself', val: '1' },
                      { label: '2 to 10 employees', val: '2-10' },
                      { label: '11 to 50 employees', val: '11-50' },
                      { label: '51 or more employees', val: '51+' },
                    ].map((item) => (
                      <button
                        key={item.val}
                        onClick={() => updateField('teamSize', item.val)}
                        className={`w-full p-4 text-left border rounded-xl flex items-center justify-between transition-all duration-300 ${
                          formData.teamSize === item.val
                            ? 'bg-neutral-100 border-neutral-100 text-neutral-950 shadow-[0_0_12px_rgba(255,255,255,0.15)]'
                            : 'bg-neutral-900/40 border-neutral-800/80 text-neutral-300 hover:border-neutral-700 hover:bg-neutral-900/60'
                        }`}
                      >
                        <span className="text-sm font-medium">{item.label}</span>
                        {formData.teamSize === item.val && (
                          <div className="w-5 h-5 rounded-full bg-neutral-950 flex items-center justify-center text-white">
                            <Check className="w-3.5 h-3.5 stroke-[3]" />
                          </div>
                        )}
                      </button>
                    ))}
                    {errors.teamSize && (
                      <p className="text-xs text-rose-500 ml-1">{errors.teamSize}</p>
                    )}
                  </div>
                </div>
              )}

              {}
              {/* Step 4: Primary Goal Selection */}
              {step === 4 && (
                <div className="space-y-6">
                  <div className="space-y-3">
                    <h1 className="text-2xl md:text-3xl font-semibold tracking-tight text-neutral-100 animate-fade-in">
                      What is your key objective?
                    </h1>
                    <p className="text-sm md:text-base text-neutral-400 leading-relaxed">
                      This focuses your system presets. You can customize, swap, or disable modules later.
                    </p>
                  </div>

                  <div className="space-y-3">
                    {[
                      { label: 'Manage projects & assign tasks', val: 'Manage projects & tasks' },
                      { label: 'Streamline client collaboration', val: 'Streamline client collaboration' },
                      { label: 'Automate invoices & client billing', val: 'Automate billing' },
                      { label: 'Centralize secure shared assets', val: 'Centralize files' },
                    ].map((item) => (
                      <button
                        key={item.val}
                        onClick={() => updateField('primaryGoal', item.val)}
                        className={`w-full p-4 text-left border rounded-xl flex items-center justify-between transition-all duration-300 ${
                          formData.primaryGoal === item.val
                            ? 'bg-neutral-100 border-neutral-100 text-neutral-950 shadow-[0_0_12px_rgba(255,255,255,0.15)]'
                            : 'bg-neutral-900/40 border-neutral-800/80 text-neutral-300 hover:border-neutral-700 hover:bg-neutral-900/60'
                        }`}
                      >
                        <span className="text-sm font-medium">{item.label}</span>
                        {formData.primaryGoal === item.val && (
                          <div className="w-5 h-5 rounded-full bg-neutral-950 flex items-center justify-center text-white">
                            <Check className="w-3.5 h-3.5 stroke-[3]" />
                          </div>
                        )}
                      </button>
                    ))}
                    {errors.primaryGoal && (
                      <p className="text-xs text-rose-500 ml-1">{errors.primaryGoal}</p>
                    )}
                  </div>
                </div>
              )}

              {}
              {/* Controller Navigation Bar */}
              <div className="pt-6 flex items-center justify-between space-x-4">
                {step > 1 ? (
                  <button
                    onClick={handleBack}
                    className="px-5 py-3.5 bg-neutral-950 border border-neutral-800 text-neutral-400 rounded-xl text-sm font-medium transition-all hover:bg-neutral-900 hover:text-neutral-200 active:scale-[0.98] flex items-center space-x-1"
                  >
                    <ChevronLeft className="w-4 h-4" />
                    <span>Back</span>
                  </button>
                ) : (
                  <div /> // Invisible spacer to balance single forward button
                )}

                <button
                  onClick={handleNext}
                  className="px-6 py-3.5 bg-white text-neutral-950 rounded-xl text-sm font-medium transition-all hover:bg-neutral-200 active:scale-[0.98] focus:outline-none flex items-center space-x-1.5 font-medium"
                >
                  <span>{step === totalSteps ? 'Complete Setup' : 'Continue'}</span>
                  <ChevronRight className="w-4 h-4" />
                </button>
              </div>
            </div>
          )}

        </div>
      </main>

      {/* Footer Area */}
      <footer className="w-full max-w-lg mx-auto px-6 py-8 flex flex-col sm:flex-row justify-between items-center text-xs text-neutral-500 border-t border-neutral-900/60">
        <div className="flex items-center space-x-1.5 mb-2 sm:mb-0">
          <ShieldCheck className="w-3.5 h-3.5 text-neutral-600" />
          <span>Enterprise-grade security standard</span>
        </div>
        <div>
          <span>Terms & Privacy · Aura Sync © 2026</span>
        </div>
      </footer>

    </div>
  );
}
