    <script>
        function trigger() {
            const container = document.getElementById('container');
            const circle = document.getElementById('circle');
            const cta = document.getElementById('cta');

            // Clear previous bits
            const extras = container.querySelectorAll('.quantum-bit');
            extras.forEach(el => el.remove());

            // Force reflow for circle
            circle.style.animation = 'none';
            void circle.offsetWidth; 
            circle.style.animation = 'quantum-fade 1s 0.5s var(--ease-expo) forwards, quantum-breathe 6s 1.5s var(--ease-breath) infinite';

            // Force reflow for CTA to restart delayed entry
            cta.style.animation = 'none';
            void cta.offsetWidth;
            cta.style.animation = 'cta-appear 1.2s 2.7s var(--ease-expo) forwards';

            // Generate bits
            for(let i=0; i<8; i++) {
                const bit = document.createElement('div');
                bit.className = 'quantum-bit';
                bit.style.width = (200 + i*20) + 'px';
                bit.style.height = (200 + i*20) + 'px';
                bit.style.animationDelay = (i * 0.05) + 's';
                container.appendChild(bit);
            }
        }

        window.onload = trigger;
    </script>
