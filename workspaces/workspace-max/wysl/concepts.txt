import React, { useState } from 'react';
import { Workflow } from 'lucide-react';

const Node = ({ icon: Icon, gradient, shadowColor, title, description, delay }) => {
  const [isHovered, setIsHovered] = useState(false);

      style={{ 
        animation: `fadeInUp 0.8s ease-out ${delay}s both`,
      }}
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      {/* The Main Node Body */}
      <div className="relative w-32 h-32 md:w-44 md:h-44 flex items-center justify-center cursor-pointer">
        
        {/* Outer Ring / Aura - Significantly tightened and reduced */}
        <div 
          className={`absolute inset-0 rounded-full transition-all duration-1000 blur-xl opacity-5 group-hover:opacity-10 group-hover:scale-95 ${gradient}`}
        />

        <div className="absolute inset-0 rounded-full border border-white/5 bg-white/[0.02] backdrop-blur-md shadow-inner" />

        {/* The Primary Gradient Sphere */}
        <div 
          className={`absolute inset-3 rounded-full shadow-2xl transition-all duration-500 overflow-hidden ${gradient} ${isHovered ? 'scale-105' : 'scale-100'}`}
          style={{
            boxShadow: isHovered 
              ? `0 10px 25px -8px ${shadowColor}55, inset 0 4px 20px rgba(255,255,255,0.2)` 
              : `0 8px 20px -5px ${shadowColor}44, inset 0 2px 12px rgba(255,255,255,0.15)`
          }}
        >
          {/* Internal Specular Highlight (The "Glass" Look) */}
          <div className="absolute top-[-30%] left-[-20%] w-[140%] h-[140%] bg-gradient-to-br from-white/40 via-transparent to-transparent pointer-events-none" />
          
          {/* Subtle Internal Pattern */}
          <div className="absolute inset-0 opacity-20 mix-blend-overlay">
            <svg width="100%" height="100%">
              <pattern id="grid" width="12" height="12" patternUnits="userSpaceOnUse">
                <circle cx="1" cy="1" r="0.6" fill="white" />
              </pattern>
              <rect width="100%" height="100%" fill="url(#grid)" />
            </svg>
          </div>

          {/* Deep Core Shadow for dimension */}
          <div className="absolute inset-0 bg-[radial-gradient(circle_at_center,transparent_30%,rgba(0,0,0,0.4)_100%)]" />

        </h3>
        <p className="mt-2 text-sm font-medium text-slate-400 leading-relaxed opacity-80 group-hover:opacity-100 transition-opacity">
          {description}
        </p>
      </div>

      {/* Connecting Line Indicator */}
      <div className="mt-6 w-px h-16 bg-gradient-to-b from-white/20 to-transparent opacity-30 group-hover:h-20 transition-all duration-500" />
    </div>

    gradient: "bg-gradient-to-tr from-rose-600 via-red-500 to-orange-500",
    shadowColor: "#e11d48",
  };

  return (
    <div className="min-h-screen bg-[#0a0a0c] flex items-center justify-center overflow-hidden">
      <style>{`
        @keyframes fadeInUp {
          from {
            opacity: 0;
            transform: translateY(40px);
          }
          to {
            opacity: 1;
            transform: translateY(0);
          }
        }
        body {
          background-color: #0a0a0c;
        }
      `}</style>
      
      <div className="max-w-7xl mx-auto">
        <Node 
          {...processHubNode} 
          delay={0}
        />
      </div>
    </div>
  );
}
