const VariableRow = ({ label, value, isSearching }) => (
  <div className={`flex justify-between py-1.5 transition-all duration-300 ${isSearching ? 'opacity-40 scale-95 blur-[1px]' : 'opacity-100 scale-100 blur-0'}`}>

const SectionHeader = ({ icon: Icon, title, onSearch, currentSearch, isSearching, resultCount }) => {
  const [isInputActive, setIsInputActive] = useState(false);
  const inputRef = useRef(null);

  useEffect(() => {
    if (isInputActive && inputRef.current) {
      inputRef.current.focus();
    }
  }, [isInputActive]);

  return (
    <div 
      className={`group flex items-center justify-between px-3 py-2 rounded-lg cursor-pointer transition-all duration-300 mb-2 ${
        isInputActive ? 'bg-[#1a1a1c] ring-1 ring-[#2dd4bf]/30' : 'hover:bg-[#1a1a1c]'
      }`}
      onClick={() => !isInputActive && setIsInputActive(true)}
    >
      <div className="flex items-center gap-3 flex-1">
        <div className={`transition-transform duration-300 ${isInputActive ? 'rotate-90 text-[#2dd4bf]' : 'text-[#2dd4bf]'}`}>
          <Icon size={18} />
        </div>
        
        {isInputActive ? (
          <input
            ref={inputRef}
            type="text"
            className="bg-transparent border-none outline-none text-white text-sm w-full font-medium placeholder:text-gray-600"
            placeholder={`Search ${title}...`}
            value={currentSearch}
            onChange={(e) => onSearch(e.target.value)}
            onBlur={() => !currentSearch && setIsInputActive(false)}
            onKeyDown={(e) => e.key === 'Enter' && setIsInputActive(false)}
          />
        ) : (
          <span className="text-white font-bold text-sm select-none">{title}</span>
        )}
      </div>

      <div className="flex items-center gap-2">
        {currentSearch && (
          <span className="text-[10px] bg-[#2dd4bf]/20 text-[#2dd4bf] px-1.5 py-0.5 rounded uppercase tracking-wider font-bold">
            {resultCount} Found
          </span>
        )}
        <div className={`transition-opacity duration-300 ${isInputActive ? 'opacity-100' : 'opacity-0 group-hover:opacity-100'}`}>
          {isInputActive ? (
            <X 
              size={14} 
              className="text-gray-500 hover:text-white" 
              onClick={(e) => {
                e.stopPropagation();
                onSearch('');
                setIsInputActive(false);
              }}
            />
          ) : (
            <Search size={14} className="text-gray-500" />
          )}
        </div>
        {!isInputActive && <ChevronDown size={16} className="text-gray-500" />}
      </div>
    </div>
  );
};
